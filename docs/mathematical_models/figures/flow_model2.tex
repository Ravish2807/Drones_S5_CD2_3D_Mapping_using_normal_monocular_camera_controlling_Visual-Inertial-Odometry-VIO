%------------------------------------------------------------------ blocks
\defblock[equal height group=r1]{b1}{\blkw}{cPlant}{1}{Nonlinear Flight Data}{drone / simulation}{%
  $\begin{gathered}
    \vx_k=\begin{bmatrix}\vp_k-\vp_{\mathrm{ref}}\\ \vv_k\\ \mathrm{Log}(R_k)\\ \vw_k\end{bmatrix}\in\R^{12},\qquad
    \vu_k=\begin{bmatrix}T_k-mg_0\\ \vtau_k\end{bmatrix}\in\R^{4}\\[2pt]
    \text{sampled at } t_k = k\,\Delta t,\quad k=1,\dots,N
  \end{gathered}$}
  {flights with a stabilising pilot $+$ random excitation}
  {state history $\{\vx_k\}$, input history $\{\vu_k\}$}

\defblock[equal height group=r1]{b2}{\blkw}{cOuter}{2}{DMDc}{system identification}{%
  $\begin{gathered}
    X=[\vx_1\cdots\vx_{N-1}],\qquad X^{+}=[\vx_2\cdots\vx_{N}]\\
    U=[\vu_1\cdots\vu_{N-1}]\\[3pt]
    [\,A\ \ B\,]=X^{+}\begin{bmatrix}X\\ U\end{bmatrix}^{\dagger}
    \ \Rightarrow\ \vx_{k+1}\approx A\vx_k+B\vu_k
  \end{gathered}$}
  {$X,X^{+}\in\R^{12\times(N-1)}$,\ \ $U\in\R^{4\times(N-1)}$}
  {$A\in\R^{12\times12}$,\ \ $B\in\R^{12\times4}$ (discrete time)}

\defblock[equal height group=r2]{b3}{\blkw}{cGuid}{3}{Bellman Equation}{optimal control}{%
  $\begin{gathered}
    J=\sum_{k=0}^{\infty}\left(\vx_k^{\top}Q\,\vx_k+\vu_k^{\top}R\,\vu_k\right)\\[2pt]
    V(\vx_k)=\min_{\vu_k}\Big[\vx_k^{\top}Q\vx_k+\vu_k^{\top}R\vu_k+V(\vx_{k+1})\Big]\\[1pt]
    V(\vx)=\vx^{\top}P\,\vx\quad\text{(quadratic value function)}
  \end{gathered}$}
  {$A,B$; weights $Q\succeq0$, $R\succ0$}
  {optimal input is linear: $\vu^{*}=-K\vx$}

\defblock[equal height group=r2]{b7}{\blkw}{cInner}{7}{Model Predictive Control}{constrained}{%
  $\begin{gathered}
    \min_{\vu_0,\dots,\vu_{N-1}}\ \sum_{k=0}^{N-1}\left(\vx_k^{\top}Q\vx_k+\vu_k^{\top}R\vu_k\right)
      +\vx_N^{\top}P_f\,\vx_N\\[1pt]
    \text{s.t.}\ \ \vx_{k+1}=A\vx_k+B\vu_k,\quad \vu_k\in\mathcal{U},\quad \vx_k\in\mathcal{X}
  \end{gathered}$}
  {$A,B,Q,R$, horizon $N$, limits $\mathcal{U},\mathcal{X}$, $P_f$}
  {finite-horizon optimal control problem}

\defblock[equal height group=r3]{b4}{\blkw}{cGuid}{4}{Linear Quadratic Regulator}{optimal control}{%
  $\begin{gathered}
    K=\left(R+B^{\top}PB\right)^{-1}B^{\top}PA\\[2pt]
    \vu_k=-K\,\vx_k
  \end{gathered}$}
  {$A,B,R$ and value matrix $P$}
  {feedback gain $K\in\R^{4\times12}$}

\defblock[equal height group=r3]{b8}{\blkw}{cInner}{8}{Quadratic Program}{stacked form}{%
  $\begin{gathered}
    \min_{\mathbf U}\ \tfrac12\mathbf U^{\top}H\mathbf U+\bm{f}^{\top}\mathbf U\quad
    \text{s.t.}\ \ G\mathbf U\le\bm{h},\ \ A_{eq}\mathbf U=\bm{b}_{eq}\\[2pt]
    H=2\big(S_u^{\top}\bar{Q}S_u+\bar{R}\big),\qquad \bm{f}=2S_u^{\top}\bar{Q}S_x\vx_0
  \end{gathered}$}
  {MPC problem, current state $\vx_0$}
  {$H\in\R^{4N\times4N}$, $\bm f$, $G$, $\bm h$ $\rightarrow$ QP solver}

\defblock[equal height group=r4]{b5}{\blkw}{cGuid}{5}{Riccati Recursion}{iterative}{%
  $P_{k+1}=A^{\top}P_kA-A^{\top}P_kB\left(R+B^{\top}P_kB\right)^{-1}B^{\top}P_kA+Q$}
  {$A,B,Q,R$; start $P_0=Q$}
  {sequence $P_0,P_1,\dots\rightarrow P_\infty$}

\defblock[equal height group=r4]{b9}{\blkw}{cInner}{9}{MPC Control Input}{receding horizon}{%
  $\mathbf U^{*}=[\,\vu_0^{*},\vu_1^{*},\dots,\vu_{N-1}^{*}\,]
   \ \longrightarrow\ \text{apply only } \vu_0^{*},\ \text{then re-solve at } k+1$}
  {QP data $(H,\bm f,G,\bm h)$ at time $k$}
  {$\vu_0^{*}\in\R^4$}

\defblock[equal height group=r5]{b6}{\blkw}{cGuid}{6}{DARE}{steady state}{%
  $P=A^{\top}PA-A^{\top}PB\left(R+B^{\top}PB\right)^{-1}B^{\top}PA+Q$}
  {$A,B,Q,R$}
  {$P\in\R^{12\times12}$, $P\succ0$ $\Rightarrow$ final $K$}

\defblock{b10}{17.6cm}{cPlant}{10}{Apply to Drone}{actuation}{%
  $\vu_k=\vu_{\mathrm{trim}}+\begin{cases}\vu_0^{*} & \text{(MPC)}\\ -K\vx_k & \text{(LQR)}\end{cases}
   =\begin{bmatrix}T_k\\ \vtau_k\end{bmatrix}
   \ \longrightarrow\ \vf=\Gamma^{-1}\vu_k
   \ \longrightarrow\ \Omega_i=\sqrt{f_i/k_f}$}
  {measured state $\vx_k$, controller output}
  {motor speeds $\vOm\in\R^4$ $\rightarrow$ the drone moves to $\vx_{k+1}$}
