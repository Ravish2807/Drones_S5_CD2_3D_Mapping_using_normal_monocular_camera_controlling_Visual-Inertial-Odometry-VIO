\newtcolorbox{flowbox}[3][]{enhanced, colback=white, colframe=#2, colbacktitle=#2,
  coltitle=white, title={#3}, fonttitle=\sffamily\bfseries\small,
  boxrule=0.8pt, arc=3pt, left=5pt, right=5pt, top=4pt, bottom=4pt,

  boxsep=0pt, left=2.6pt, right=2.6pt, top=0.6pt, bottom=0.6pt,
  fontupper=\sffamily\bfseries\small]{#2}}
\newcommand{\ltag}[1]{\hfill{\mdseries\scriptsize\MakeUppercase{#1}}}
\newcommand{\io}[2]{\textbf{IN}\enspace #1\par\smallskip\textbf{OUT}\enspace #2}

\makeatletter
\newcommand{\defblock}[9][]{\def\db@opts{#1}\def\db@name{#2}\def\db@w{#3}\def\db@c{#4}%
  \def\db@n{#5}\def\db@t{#6}\def\db@tag{#7}\def\db@body{#8}\def\db@in{#9}\db@cont}
\newcommand{\db@cont}[1]{%
  \expandafter\newsavebox\csname blk@\db@name\endcsname
  \expandafter\sbox\csname blk@\db@name\endcsname{%
    \begin{minipage}{\db@w}%
      \edef\db@exp{\noexpand\begin{flowbox}[\db@opts]{\db@c}}\db@exp
        {\num{\db@c}{\db@n}\ \db@t\ltag{\db@tag}}%
        \centering\db@body
        \tcblower\raggedright\io{\db@in}{#1}%
      \end{flowbox}%
    \end{minipage}}}
\makeatother
\newcommand{\useblock}[1]{\expandafter\usebox\csname blk@#1\endcsname}
