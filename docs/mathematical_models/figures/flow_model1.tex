\newlength{\blkw}\setlength{\blkw}{6.7cm}
\defblock[equal height group=m1top]{b1}{\blkw}{cPlant}{1}{Nonlinear Dynamics}{plant}{%
  $\begin{aligned}
    \dot{\vp} &= \vv\\
    \dot{\vv} &= \vg + \tfrac{T}{m}\,R\,\ve_3\\
    \dot{R}   &= R\,\skw{\vw}\\
    \dot{\vw} &= J^{-1}\!\left(\vtau - \vw\times J\vw\right)
  \end{aligned}$}
  {$\vu=[T;\vtau]\in\R^{4}$\ \ (N;\ N\,m)}
  {$\vx=(\vp,\vv,R,\vw)$:\ \ $\R^3{\times}\R^3{\times}\SO{\times}\R^3$}

\defblock[equal height group=m1top]{b2}{\blkw}{cModel}{2}{Linearization}{model}{%
  $\begin{gathered}
    \vf(\vx,\vu)\approx\vf(\hx,\hat{\vu}) + A\,\delta\vx + B\,\delta\vu\\[2pt]
    A=\left.\dfrac{\partial\vf}{\partial\vx}\right|_{\hx,\hat{\vu}}\qquad
    B=\left.\dfrac{\partial\vf}{\partial\vu}\right|_{\hx,\hat{\vu}}\\[2pt]
    \delta\vu=\vu-\hat{\vu},\qquad R=\hR\,\mathrm{Exp}(\delta\bm\theta)
  \end{gathered}$}
  {operating point $(\hx,\hat{\vu})$, model $\vf$}
  {$A\in\R^{12\times12}$,\ \ $B\in\R^{12\times4}$}

\defblock[equal height group=m1top]{b3}{\blkw}{cModel}{3}{Local Error Dynamics}{model}{%
  $\begin{gathered}
    \delta\vx=\begin{bmatrix}\delta\vp\\ \delta\vv\\ \delta\bm\theta\\ \delta\vw\end{bmatrix}
      =\begin{bmatrix}\vp-\hp\\ \vv-\hv\\ \mathrm{Log}(\hR^{\top}R)\\ \vw-\hw\end{bmatrix}\in\R^{12}\\[3pt]
    \delta\dot{\vx} = A\,\delta\vx + B\,\delta\vu\\[2pt]
    F = I + A\,\Delta t \;\approx\; e^{A\Delta t}
  \end{gathered}$}
  {$A$, $B$, sample time $\Delta t$}
  {$F\in\R^{12\times12}$ (error transition)}

\defblock[equal height group=m1top]{b4}{\blkw}{cEst}{4}{Error-State EKF}{estimation}{%
  {\scriptsize\textbf{Prediction}}\\[-1pt]
  $\begin{aligned}
    \hx_{k+1} &= \vf_d(\hx_k,\vu_k)\\
    P_{k+1}   &= F P_k F^{\top} + Q
  \end{aligned}$\\[3pt]
  {\scriptsize\textbf{Correction}}\\[-1pt]
  $\begin{aligned}
    K   &= P H^{\top}\!\left(H P H^{\top} + R_n\right)^{-1}\\
    \delta\hat{\vx} &= K\left(\vz - h(\hx)\right)\\
    \hp,\hv,\hw &\leftarrow \hp{+}\delta\hat{\vp},\ \hv{+}\delta\hat{\vv},\ \hw{+}\delta\hat{\vw}\\
    \hR &\leftarrow \hR\,\mathrm{Exp}(\delta\hat{\bm\theta})\\
    P   &\leftarrow (I - K H)\,P
  \end{aligned}$}
  {$\vu_k$ (applied), $\vz_k$ (sensors), $F$}
  {$\hx_k=(\hp,\hv,\hR,\hw)$,\ \ $P_k\in\R^{12\times12}$}

\defblock[equal height group=m1bot]{b5}{\blkw}{cOuter}{5}{Position Control}{outer loop}{%
  $\begin{gathered}
    \ve_p = \vp_d - \hp \\
    \ve_v = \vv_d - \hv\\[3pt]
    \va_d = \ddot{\vp}_d + K_p\,\ve_p + K_v\,\ve_v
  \end{gathered}$}
  {$\hp,\hv$;\ \ $\vp_d,\vv_d,\ddot{\vp}_d$}
  {$\va_d\in\R^{3}$ (m/s$^2$)}

\defblock[equal height group=m1bot]{b6}{\blkw}{cGuid}{6}{Force \& Attitude}{guidance}{%
  $\begin{gathered}
    \vF_d = m\,(\va_d - \vg), \qquad \vb_{3d} = \dfrac{\vF_d}{\|\vF_d\|}\\[3pt]
    \vb_{2d}=\dfrac{\vb_{3d}\times\vb_{1c}}{\|\vb_{3d}\times\vb_{1c}\|},\qquad
    \vb_{1d}=\vb_{2d}\times\vb_{3d}\\[2pt]
    R_d=[\,\vb_{1d}\ \ \vb_{2d}\ \ \vb_{3d}\,]
  \end{gathered}$}
  {$\va_d$, desired yaw $\psi_d$}
  {$\vF_d\in\R^3$ (N),\ \ $R_d\in\SO$}

\defblock[equal height group=m1bot]{b7}{\blkw}{cInner}{7}{SO(3) Attitude Control}{inner loop}{%
  $\begin{aligned}
    \ve_R &= \tfrac12\big(R_d^{\top}\hR-\hR^{\top}R_d\big)^{\vee}\\
    \ve_\omega &= \hw-\hR^{\top}R_d\,\vw_d\\
    \vtau &= -K_R\ve_R-K_\omega\ve_\omega+\hw\times J\hw\\
          &\quad -J\big(\skw{\hw}\hR^{\top}R_d\vw_d-\hR^{\top}R_d\dot{\vw}_d\big)\\
    T &= \vF_d\cdot\hR\,\ve_3
  \end{aligned}$}
  {$\hR,\hw$;\ \ $R_d,\vF_d$;\ \ $\vw_d,\dot{\vw}_d$}
  {$T\in\R$ (N),\ \ $\vtau\in\R^3$ (N\,m)}

\defblock[equal height group=m1bot]{b8}{\blkw}{cPlant}{8}{Control Allocation}{actuators}{%
  $\begin{gathered}
    \begin{bmatrix}T\\ \vtau\end{bmatrix}=\Gamma\,\vf,\qquad f_i = k_f\,\Omega_i^2\\[3pt]
    \Omega_i=\sqrt{\big(\Gamma^{-1}\vu\big)_i/k_f}\\
    \text{(clipped to } [\Omega_{\min},\Omega_{\max}])
  \end{gathered}$}
  {$\vu=[T;\vtau]$, rotor geometry $\Gamma$, $k_f$}
  {motor speeds $\vOm\in\R^4$ (rad/s)}
